3) Add a simple voice hook (hooks/useVoice.ts) – STT + TTS

If a voice hook already exists from previous iterations, either reuse or replace it with a clean, minimal version that never touches the layout.

Create/update src/hooks/useVoice.ts:

import { useCallback, useEffect, useState } from "react";

interface VoiceState {
  hasSpeechSupport: boolean;
  hasTTSSupport: boolean;
  isListening: boolean;
  isSpeaking: boolean;
  lastTranscript: string;
  error: string | null;
  startListening: () => void;
  stopListening: () => void;
  speak: (text: string) => void;
  clearTranscript: () => void;
}

export function useVoice(): VoiceState {
  const [hasSpeechSupport, setHasSpeechSupport] = useState(false);
  const [hasTTSSupport, setHasTTSSupport] = useState(false);
  const [isListening, setIsListening] = useState(false);
  const [isSpeaking, setIsSpeaking] = useState(false);
  const [lastTranscript, setLastTranscript] = useState("");
  const [error, setError] = useState<string | null>(null);

  const [recognition, setRecognition] = useState<SpeechRecognition | null>(null);

  useEffect(() => {
    const SpeechRecognitionCtor: any =
      (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
    if (SpeechRecognitionCtor) {
      const rec: SpeechRecognition = new SpeechRecognitionCtor();
      rec.continuous = false;
      rec.interimResults = false;
      rec.lang = "en-US";
      rec.onresult = event => {
        const text = event.results[0][0].transcript;
        setLastTranscript(text);
      };
      rec.onerror = ev => {
        setError(`Mic error: ${ev.error}`);
        setIsListening(false);
      };
      rec.onend = () => setIsListening(false);
      setRecognition(rec);
      setHasSpeechSupport(true);
    }
    if ("speechSynthesis" in window) {
      setHasTTSSupport(true);
    }
  }, []);

  const startListening = useCallback(() => {
    if (!recognition) {
      setError("Voice input not supported in this browser.");
      return;
    }
    setError(null);
    setLastTranscript("");
    try {
      recognition.start();
      setIsListening(true);
    } catch {
      // ignore if already started
    }
  }, [recognition]);

  const stopListening = useCallback(() => {
    if (recognition && isListening) {
      recognition.stop();
    }
  }, [recognition, isListening]);

  const speak = useCallback(
    (text: string) => {
      if (!hasTTSSupport || !text) return;
      const synth = window.speechSynthesis;
      synth.cancel();
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.rate = 1;
      utterance.pitch = 1.05;
      utterance.onstart = () => setIsSpeaking(true);
      utterance.onend = () => setIsSpeaking(false);
      synth.speak(utterance);
    },
    [hasTTSSupport]
  );

  return {
    hasSpeechSupport,
    hasTTSSupport,
    isListening,
    isSpeaking,
    lastTranscript,
    error,
    startListening,
    stopListening,
    speak,
    clearTranscript: () => setLastTranscript("")
  };
}


Use standard SpeechRecognition and speechSynthesis browser APIs. Typescript complaints about SpeechRecognition can be fixed by declaring it as any at the top if needed.
